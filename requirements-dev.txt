# Host-side tooling for Robox firmware development.
# Nothing here is installed on the Pico.
#
#   python3 -m pip install --user -r requirements-dev.txt

# Official MicroPython remote-control CLI. Backs every task in
# .vscode/tasks.json via tools/pico.
mpremote>=1.24.0

# littlefs image builder, used by tools/build_uf2.py to assemble a release UF2
# on the host instead of dumping one off a board. The Pico's filesystem is
# littlefs2, so this is what lets `./tools/pico build` work with no hardware.
littlefs-python>=0.12

#

#
